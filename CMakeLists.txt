cmake_minimum_required(VERSION 3.29)
project(AutoDecoStudio VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUTODECO_TEST_ONLY "Build pure C++ generator tests without Geode SDK" OFF)
if(AUTODECO_TEST_ONLY)
    enable_testing()
    add_executable(autodeco_tests tests/planner_tests.cpp)
    add_test(NAME AutoDecoTests COMMAND autodeco_tests)
    return()
endif()
add_library(${PROJECT_NAME} SHARED src/main.cpp)
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "GEODE_SDK not set - use Geode CLI or GitHub Actions")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME})
